import { S3Client, PutObjectCommand } from "npm:@aws-sdk/client-s3@3"; import { createClient } from "npm:@supabase/supabase-js@2"; const corsHeaders = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Headers": "authorization, x-client-info, apikey, content-type", "Access-Control-Allow-Methods": "POST, OPTIONS" }; const SUPABASE_URL = Deno.env.get("SUPABASE_URL")!; const SUPABASE_ANON_KEY = Deno.env.get("SUPABASE_ANON_KEY")!; const R2_ACCOUNT_ID = Deno.env.get("R2_ACCOUNT_ID")!; const R2_ACCESS_KEY_ID = Deno.env.get("R2_ACCESS_KEY_ID")!; const R2_SECRET_ACCESS_KEY = Deno.env.get("R2_SECRET_ACCESS_KEY")!; const R2_BUCKET_NAME = Deno.env.get("R2_BUCKET_NAME")!; const R2_PUBLIC_URL = (Deno.env.get("R2_PUBLIC_URL") || "").replace(/\/$/, ""); const MAX_BYTES = 20 * 1024 * 1024; const supabase = createClient(SUPABASE_URL, SUPABASE_ANON_KEY); const r2 = new S3Client({ region: "auto", endpoint: `https://${R2_ACCOUNT_ID}.r2.cloudflarestorage.com`, credentials: { accessKeyId: R2_ACCESS_KEY_ID, secretAccessKey: R2_SECRET_ACCESS_KEY } }); function json(body: unknown, status=200){ return new Response(JSON.stringify(body),{ status, headers:{...corsHeaders,"Content-Type":"application/json"} }); } function extensionFor(assetType:string){ return assetType==="usdz" ? "usdz" : "glb"; } function planAllows3D(plan:any){ const key=`${plan?.slug||""} ${plan?.name||""}`.toLowerCase(); return Number(plan?.signature_3d_dishes_max||0)>0 && ( key.includes("signature 3d") || key.includes("signature-3d") || key.includes("signature_3d") || (key.includes("signature") && key.includes("3d")) || key.includes("full 3d") || key.includes("full-3d") ); } Deno.serve(async(req)=>{ if(req.method==="OPTIONS")return new Response("ok",{headers:corsHeaders}); if(req.method!=="POST")return json({error:"POST required"},405); try{ if(!R2_ACCOUNT_ID||!R2_ACCESS_KEY_ID||!R2_SECRET_ACCESS_KEY||!R2_BUCKET_NAME||!R2_PUBLIC_URL){ return json({error:"R2 secrets are not configured."},500); } const authHeader=req.headers.get("Authorization")||""; if(!authHeader.startsWith("Bearer "))return json({error:"Authentication required."},401); const token=authHeader.replace(/^Bearer\s+/i,""); const {data:userData,error:userError}=await supabase.auth.getUser(token); if(userError||!userData.user)return json({error:"Invalid or expired login session."},401); const form=await req.formData(); const file=form.get("file"); const restaurantId=String(form.get("restaurant_id")||"").trim(); const menuItemId=String(form.get("menu_item_id")||"").trim(); const assetType=String(form.get("asset_type")||"glb").trim().toLowerCase(); if(!(file instanceof File))return json({error:"3D model file is required."},400); if(!restaurantId)return json({error:"restaurant_id is required."},400); if(!menuItemId)return json({error:"menu_item_id is required."},400); if(!["glb","usdz"].includes(assetType))return json({error:"Only GLB or USDZ is allowed."},400); if(file.size>MAX_BYTES)return json({error:"3D model must be 20 MB or smaller."},400); const extension=(file.name.split(".").pop()||"").toLowerCase(); if(extension!==assetType){ return json({error:`Please upload a .${assetType} file.`},400); } const {data:restaurant,error:restaurantError}=await supabase .from("restaurants") .select("id,owner_id,plan_id") .eq("id",restaurantId) .maybeSingle(); if(restaurantError)return json({error:restaurantError.message},500); if(!restaurant)return json({error:"Restaurant not found."},404); let allowed=restaurant.owner_id===userData.user.id; if(!allowed){ const {data:profile}=await supabase .from("profiles") .select("role") .eq("id",userData.user.id) .maybeSingle(); allowed=profile?.role==="admin"; } if(!allowed)return json({error:"You are not allowed to upload for this restaurant."},403); const {data:plan,error:planError}=await supabase .from("plans") .select("id,name,slug,is_active,signature_3d_dishes_max") .eq("id",restaurant.plan_id) .maybeSingle(); if(planError)return json({error:planError.message},500); if(!plan||plan.is_active===false||!planAllows3D(plan)){ return json({error:"Signature 3D or Full 3D Experience plan is required."},403); } const {data:item,error:itemError}=await supabase .from("menu_items") .select("id,restaurant_id") .eq("id",menuItemId) .eq("restaurant_id",restaurantId) .maybeSingle(); if(itemError)return json({error:itemError.message},500); if(!item)return json({error:"Menu item not found for this restaurant."},404); const objectKey= `restaurants/${restaurantId}/signature-3d/${menuItemId}/${crypto.randomUUID()}.${extensionFor(assetType)}`; await r2.send(new PutObjectCommand({ Bucket:R2_BUCKET_NAME, Key:objectKey, Body:new Uint8Array(await file.arrayBuffer()), ContentType:assetType==="glb" ? "model/gltf-binary" : "model/vnd.usdz+zip", CacheControl:"public, max-age=31536000, immutable" })); return json({ success:true, url:`${R2_PUBLIC_URL}/${objectKey}`, object_key:objectKey, restaurant_id:restaurantId, menu_item_id:menuItemId, asset_type:assetType, storage_provider:"cloudflare-r2" }); }catch(error){ console.error("signature-3d-model-upload error",error); return json({ error:error instanceof Error?error.message:"3D R2 upload failed." },500); } });
